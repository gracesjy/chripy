def click_with_alert_handling(driver, locator_tuple, alert_keywords_for_dismiss=None, default_action="accept"):
    try:
        element = driver.find_element(*locator_tuple)
        element.click()
    except UnexpectedAlertPresentException:
        handle_unexpected_alert(driver, alert_keywords_for_dismiss, default_action)
        element = driver.find_element(*locator_tuple)
        element.click()

def handle_unexpected_alert(driver, alert_keywords_for_dismiss=None, default_action="accept"):
    try:
        alert = driver.switch_to.alert
        text = alert.text
        print("[Alert Detected]", text)
        if alert_keywords_for_dismiss and any(kw in text for kw in alert_keywords_for_dismiss):
            print("[Alert Action] dismiss()")
            alert.dismiss()
        else:
            if default_action == "accept":
                print("[Alert Action] accept()")
                alert.accept()
            else:
                print("[Alert Action] dismiss()")
                alert.dismiss()
    except NoAlertPresentException:
        pass
